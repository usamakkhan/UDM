struct AutomaticProbe {

 static void CALLBACK tick(HWND,UINT,UINT_PTR,DWORD){auto* self=active;if(!self||self->seen)return;HWND dialog=nullptr;
  EnumThreadWindows(GetCurrentThreadId(),[](HWND window,LPARAM data)->BOOL{wchar_t title[256]{};GetWindowTextW(window,title,256);if(IsWindowVisible(window)&&GetWindow(window,GW_OWNER)==active->owner&&!wcscmp(title,L"UDM automatic presentation acceptance")){*reinterpret_cast<HWND*>(data)=window;return FALSE;}return TRUE;},reinterpret_cast<LPARAM>(&dialog));
  if(!dialog){if(GetTickCount64()-self->began<7000)return;self->seen=true;self->error="Owned blocker did not appear";return;}
  self->seen=true;try{self->inspect();}catch(const std::exception& e){self->error=e.what();}catch(...){self->error="Owned automatic-presentation observation failed";}PostMessageW(dialog,WM_CLOSE,0,0);
 }
 AutomaticProbe(HWND parent,std::function<void()> action):owner(parent),inspect(std::move(action)){active=this;timer=SetTimer(nullptr,0,10,tick);if(!timer)throw std::runtime_error("Cannot observe owned automatic blocker");}

};
