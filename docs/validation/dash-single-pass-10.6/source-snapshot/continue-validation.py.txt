from pathlib import Path
import json,subprocess,os
repo=Path('D:/UDM');area=repo/'.media-cache/dash-single-pass-build-10.6';release=area/'release-native'
# Recipes are pinned before their runtime; no commands are loaded from evidence.
recipes=['tests/dash-single-pass.native-live.py','tests/dash-metadata-transport.native-live.py','tests/dash-metadata-cancellation.native-live.py','tests/dash-freshness.native-live.py','tests/dash-freshness-clock.native-live.py','tests/live-dash-continuous.native-live.py','tests/gapless-aac-tail.native-live.py']
import hashlib
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest().upper()
pins=[dict(path=p,sha256=sha(repo/p)) for p in recipes]
(area/'runtime-source-pins.json').write_text(json.dumps(pins,indent=2),encoding='utf-8')
env=os.environ.copy();env.update(TEMP=str(area),TMP=str(area),UDM_CONTINUOUS_DASH_TEST_EXE=str(release/'Udm.ContinuousDashWorkflow.exe'))
def run(args,name):
 with (area/(name+'.log')).open('w',encoding='utf-8') as log:r=subprocess.run(list(map(str,args)),cwd=repo,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=500)
 print(name+': '+str(r.returncode),flush=True)
 if r.returncode:print((area/(name+'.log')).read_text(encoding='utf-8',errors='replace')[-4500:],flush=True)
 assert r.returncode==0,name
for name in ['DashMetadataTransportTests','GaplessAacTailTests','RetiredDashBoundaryTests']:
 run([release/('Udm.'+name+'.exe'),area/('qualified-'+name)],name)
run(['python',repo/'tests/live-dash-continuous.native-live.py',area/'continuous-qualified'],'continuous-qualified')
for name,control,extra in [('direct-qualified','10.0',[]),('caption-qualified','10.1',['--caption-sessions'])]:
 run(['python',repo/'tests/gapless-aac-tail.native-live.py',area/name,release/'Udm.GaplessTailWorkflow.exe',release/('Udm.GaplessTailWorkflow-Control'+control+'.exe'),*extra],name)
assert all(sha(repo/e['path'])==e['sha256'] for e in pins)
(area/'runtime-complete.json').write_text(json.dumps(dict(passed=True,sourcePins=pins),indent=2),encoding='utf-8')
